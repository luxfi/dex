## Overview

The bridge supports:
- **Multi-chain transfers** across major blockchains
- **Lock-and-mint** mechanism for wrapped assets
- **MPC threshold signatures** for security
- **Sub-minute finality** for transfers

## Bridge Architecture

## Bridge Architecture

### Lock Contract (Source Chain)

```solidity
// Simplified Ethereum bridge contract
contract LuxBridge {
    mapping(bytes32 => bool) public processedTransfers;
    address[] public validators;
    uint256 public threshold = 3;
    
    function lock(
        address token,
        uint256 amount,
        string calldata destAddress
    ) external payable {
        // Transfer tokens to bridge
        IERC20(token).transferFrom(msg.sender, address(this), amount);
        
        emit TokensLocked(
            msg.sender,
            token,
            amount,
            destAddress,
            block.timestamp
        );
    }
    
    function unlock(
        address token,
        uint256 amount,
        address recipient,
        bytes[] calldata signatures
    ) external {
        bytes32 hash = keccak256(abi.encodePacked(token, amount, recipient));
        require(!processedTransfers[hash], "Already processed");
        require(verifySignatures(hash, signatures), "Invalid signatures");
        
        processedTransfers[hash] = true;
        IERC20(token).transfer(recipient, amount);
    }
}
```

## Transfer Status

Track transfer progress:

```go
// Poll for status
for {
    status, err := bridge.GetTransferStatus(ctx, transferID)
    
    switch status.Status {
    case "pending":
        fmt.Println("Waiting for source confirmation...")
    case "confirming":
        fmt.Printf("Confirmations: %d/%d\n", 
            status.Confirmations, status.Required)
    case "signing":
        fmt.Println("MPC validators signing...")
    case "relaying":
        fmt.Println("Relaying to destination chain...")
    case "completed":
        fmt.Printf("Transfer complete! Tx: %s\n", status.DestTxHash)
        return
    case "failed":
        fmt.Printf("Transfer failed: %s\n", status.Error)
        return
    }
    
    time.Sleep(5 * time.Second)
}
```

## Events

```go
// Listen for bridge events
bridge.OnTransferInitiated(func(event *TransferEvent) {
    fmt.Printf("Transfer initiated: %s\n", event.ID)
    fmt.Printf("From: %s (%s)\n", event.SourceChain, event.SourceAddress)
    fmt.Printf("Amount: %s %s\n", event.Amount, event.Asset)
})

bridge.OnTransferCompleted(func(event *TransferEvent) {
    fmt.Printf("Transfer completed: %s\n", event.ID)
    fmt.Printf("Destination tx: %s\n", event.DestTxHash)
})

bridge.OnTransferFailed(func(event *TransferEvent) {
    fmt.Printf("Transfer failed: %s\n", event.ID)
    fmt.Printf("Reason: %s\n", event.Error)
})
```

## Troubleshooting

### Transfer Stuck

If transfer is stuck in "confirming" state:

1. Check source chain confirmations
2. Verify bridge validators are online
3. Contact support with transfer ID

### Insufficient Balance

```go
// Check bridge liquidity before transfer
liquidity, err := bridge.GetLiquidity(ctx, "ethereum", "ETH")

if liquidity.Available < transferAmount {
    fmt.Printf("Insufficient liquidity: %s available\n", liquidity.Available)
}
```

### Wrong Network

Always verify network before signing:

```go
// Verify chain ID
chainInfo, err := bridge.GetChainInfo(ctx, "ethereum")

if chainInfo.ChainID != 1 {
    return fmt.Errorf("wrong network: expected mainnet")
}
```
